// Throwaway proof of the new handleMobileMenu, run from the template with
// --config pointing at this file's directory. Not part of the repo.
import { test, expect } from '@playwright/test';
import { handleMobileMenu } from '/Users/kskinner/QA-framework-template/test-helpers.js';

const BASE = 'http://127.0.0.1:8765/';
const sel = (toggle, menu) => ({ mobileToggle: p => p.locator(toggle), mobileMenu: p => p.locator(menu) });

test('opens a late-wired, off-screen menu, and leaves it open', async ({ page }) => {
  await page.goto(BASE);
  // Off-screen counts as "visible" to Playwright: the reason for toBeInViewport.
  await expect(page.locator('#menu')).toBeVisible();
  await expect(page.locator('#menu')).not.toBeInViewport();
  const opened = await handleMobileMenu(page, sel('#toggle', '#menu'), 'PROOF');
  expect(opened).toBe(true);
  await expect(page.locator('#menu')).toBeInViewport();
});

test('a menu that never opens fails instead of being logged and skipped', async ({ page }) => {
  await page.goto(BASE);
  await page.waitForTimeout(2000); // after the page has wired its own button
  await page.evaluate(() => document.getElementById('toggle').replaceWith(Object.assign(document.createElement('button'), { id: 'toggle', textContent: '☰' })));
  await expect(handleMobileMenu(page, sel('#toggle', '#menu'), 'PROOF')).rejects.toThrow();
});

test('no menu button: returns false and does nothing', async ({ page }) => {
  await page.goto(BASE);
  expect(await handleMobileMenu(page, sel('#no-such-button', '#menu'), 'PROOF')).toBe(false);
});
